      SELECT o."OrderID", o."ShipCountry"
      FROM "Orders" o
      WHERE EXISTS (SELECT 1 FROM "Order Details" od
                    WHERE od."OrderID" = o."OrderID"
                    HAVING sum(od."Quantity") > 50)
      ORDER BY o."OrderID";
      -> 23 UK · 25 Germany · 43 Austria · 45 Germany · 63 Germany · +1
